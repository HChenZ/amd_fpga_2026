\newcommand{\placefigure}[3]{%
  \sbox{\figdescbox}{\if\relax\detokenize{#3}\relax\else
    \begin{minipage}{0.96\linewidth}\footnotesize\color{black!70}#3\end{minipage}\fi}%
  \sbox{\figcapbox}{\parbox{\linewidth}{\centering\figcaptionpart{#1}{#2}}}%
  \setlength{\fignat}{\dimexpr\ht\figplacebox+\dp\figplacebox\relax}%
  % 图题盒高度 + 图与图题之间的行间胶与取整余量
  \setlength{\figcap}{\dimexpr\ht\figcapbox+\dp\figcapbox+4pt\relax}%
  \ifdim\dimexpr\fignat+\figcap\relax>0.9\textheight
    \sbox{\figplacebox}{\resizebox*{!}{\dimexpr0.9\textheight-\figcap\relax}{\usebox{\figplacebox}}}%
    \setlength{\fignat}{\dimexpr\ht\figplacebox+\dp\figplacebox\relax}%
  \fi
  % 页顶时段前间距会被丢弃；页中扣除段前 \intextsep 与上一行深度
  \ifdim\pagegoal=\maxdimen \setlength{\figrem}{\textheight}%
  \else \setlength{\figrem}{\dimexpr\pagegoal-\pagetotal-\pagedepth-\intextsep\relax}\fi
  \setlength{\figfit}{\dimexpr\figrem-\figcap\relax}%
  \figcheckqueue
  \iffigqueued
    \typeout{FIGPLACE #1: float-queued rem=\the\figrem\space nat=\the\fignat\space cap=\the\figcap}%
    \figfloat
  \else\ifdim\fignat<\figfit
    \typeout{FIGPLACE #1: here rem=\the\figrem\space nat=\the\fignat\space cap=\the\figcap}%
    \fighere
  \else\ifdim\figfit>0.62\fignat
    \typeout{FIGPLACE #1: scaled rem=\the\figrem\space nat=\the\fignat\space cap=\the\figcap}%
    \sbox{\figplacebox}{\resizebox*{!}{\figfit}{\usebox{\figplacebox}}}%
    \fighere
  \else
    \typeout{FIGPLACE #1: float rem=\the\figrem\space nat=\the\fignat\space cap=\the\figcap}%
    \figfloat
  \fi\fi\fi}
